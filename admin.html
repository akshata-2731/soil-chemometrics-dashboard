Default Logic — Source Code
// Weighted Similarity Distance Algorithm // Based on: S = 1 / (D + 1) where D = sum(|sample[i] - ref[i]|) // for each selected wavelength index. // Weights normalized so W_min + W_mid + W_max = 1. // Prediction = vals[0]*W_min + vals[1]*W_mid + vals[2]*W_max function predictNutrient(nutrientData, sampleValues) { const { vals, indices, min, mid, max } = nutrientData; const sel = indices; const Dmin = sel.reduce((s,i) => s + Math.abs(sampleValues[i]-(min[i]||0)), 0); const Dmid = sel.reduce((s,i) => s + Math.abs(sampleValues[i]-(mid[i]||0)), 0); const Dmax = sel.reduce((s,i) => s + Math.abs(sampleValues[i]-(max[i]||0)), 0); const Smin = 1/(Dmin+1), Smid = 1/(Dmid+1), Smax = 1/(Dmax+1); const total = Smin + Smid + Smax; return (vals[0]*Smin + vals[1]*Smid + vals[2]*Smax) / total; }
